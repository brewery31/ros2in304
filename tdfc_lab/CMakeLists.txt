cmake_minimum_required(VERSION 3.16)
project(tdfc_lab LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/Export.h" "#define RVO3D_EXPORT\n")
file(GLOB RVO_SOURCES CONFIGURE_DEPENDS "third_party/RVO2-3D/src/*.cc")
add_library(tdfc_orca SHARED tdfc_lab/research/orca_bridge.cc ${RVO_SOURCES})
target_compile_features(tdfc_orca PRIVATE cxx_std_11)
target_include_directories(tdfc_orca PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" third_party/RVO2-3D/src)
set_target_properties(tdfc_orca PROPERTIES OUTPUT_NAME orca SUFFIX ".so")
install(TARGETS tdfc_orca LIBRARY DESTINATION share/${PROJECT_NAME}/lib)
install(DIRECTORY config docs DESTINATION share/${PROJECT_NAME})
install(FILES SOURCE_MANIFEST.json README.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY third_party/RVO2-3D/LICENSES DESTINATION share/${PROJECT_NAME}/third_party/RVO2-3D)
install(FILES third_party/RVO2-3D/LICENSE DESTINATION share/${PROJECT_NAME}/third_party/RVO2-3D)
install(PROGRAMS scripts/tdfc_observe scripts/tdfc_offline DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(adapter test/test_adapter.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  ament_add_pytest_test(research_interface test/test_research_interface.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  ament_add_pytest_test(archived_replay test/test_archived_replay.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
endif()
ament_package()
